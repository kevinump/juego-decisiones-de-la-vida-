    function createBoard() {
      const boardElement = document.getElementById("board");
      boardElement.innerHTML = "";

      for (let row = 9; row >= 0; row--) {
        const isEvenRow = row % 2 === 0;
        for (let col = 0; col < 10; col++) {
          let cellNumber = isEvenRow ? (row * 10 + col + 1) : (row * 10 + (9 - col) + 1);

          const cell = document.createElement("div");
          cell.className = "cell";
          cell.id = `cell-${cellNumber}`;
          
          const numberSpan = document.createElement("span");
          numberSpan.innerText = cellNumber;
          cell.appendChild(numberSpan);

          if (decisions[cellNumber]) {
            cell.classList.add("decision");
            cell.innerHTML += "<span>❓</span>";
          }

          boardElement.appendChild(cell);
        }
      }

      const playerToken = document.createElement("div");
      playerToken.id = "player-token";
      playerToken.className = "player-token";
      document.getElementById("board").appendChild(playerToken);

      updateTokenPosition();
    }

    function updateTokenPosition() {
      const cell = document.getElementById(`cell-${playerPosition}`);
      const token = document.getElementById("player-token");
      
      const cellRect = cell.getBoundingClientRect();
      const boardRect = document.getElementById("board").getBoundingClientRect();

      const top = cellRect.top - boardRect.top + (cellRect.height / 2);
      const left = cellRect.left - boardRect.left + (cellRect.width / 2);

      token.style.top = `${top}px`;
      token.style.left = `${left}px`;

      document.getElementById("status").innerText = `Posición actual: Casilla ${playerPosition}`;
    }

    const diceFaces = ["⚀", "⚁", "⚂", "⚃", "⚄", "⚅"];

    async function playTurn() {
      if (isMoving) return;
      isMoving = true;

      document.getElementById("roll-btn").disabled = true;
      document.getElementById("options-container").innerHTML = "";

      const diceElement = document.getElementById("dice");
      let diceValue = 1;
      for (let i = 0; i < 8; i++) {
        diceValue = Math.floor(Math.random() * 6) + 1;
        diceElement.innerText = diceFaces[diceValue - 1];
        await new Promise(r => setTimeout(r, 40));
      }

      let targetPosition = playerPosition + diceValue;
      if (targetPosition > TOTAL_CELLS) targetPosition = TOTAL_CELLS;

      document.getElementById("event-title").innerText = `🎲 Sacaste un ${diceValue}`;
      document.getElementById("event-desc").innerText = `Avanzando ${diceValue} casilla(s)...`;

      while (playerPosition < targetPosition) {
        playerPosition++;
        updateTokenPosition();
        await new Promise(r => setTimeout(r, 180));
      }

      if (decisions[playerPosition]) {
        presentDecision(decisions[playerPosition]);
      } else if (playerPosition === TOTAL_CELLS) {
        document.getElementById("event-title").innerText = "🏆 ¡VICTORIA!";
        document.getElementById("event-desc").innerText = "¡Llegaste a la casilla 100 con éxito!";
      } else {
        document.getElementById("roll-btn").disabled = false;
        isMoving = false;
      }
    }

    function presentDecision(decData) {
      document.getElementById("event-title").innerText = "❓ Toma una Decisión";
      document.getElementById("event-desc").innerText = decData.question;

      const optsContainer = document.getElementById("options-container");
      optsContainer.innerHTML = "";

      decData.options.forEach(opt => {
        const btn = document.createElement("button");
        btn.className = "option-btn";
        btn.innerText = opt.text;
        btn.onclick = () => handleChoice(opt);
        optsContainer.appendChild(btn);
      });
    }

    async function handleChoice(selectedOption) {
      document.getElementById("options-container").innerHTML = "";
      document.getElementById("event-title").innerText = selectedOption.correct ? " Decision Correcta" : " Decision Incorrecta";
      document.getElementById("event-desc").innerText = selectedOption.msg;

      let targetPosition = playerPosition + selectedOption.move;
      if (targetPosition < 1) targetPosition = 1;
      if (targetPosition > TOTAL_CELLS) targetPosition = TOTAL_CELLS;

      const step = selectedOption.move > 0 ? 1 : -1;
      while (playerPosition !== targetPosition) {
        playerPosition += step;
        updateTokenPosition();
        await new Promise(r => setTimeout(r, 200));
      }

      if (playerPosition === TOTAL_CELLS) {
        document.getElementById("event-title").innerText = "🏆 ¡VICTORIA!";
        document.getElementById("event-desc").innerText = "¡Has ganado el juego!";
      } else {
        document.getElementById("roll-btn").disabled = false;
        isMoving = false;
      }
    }

    function resetGame() {
      playerPosition = 1;
      updateTokenPosition();
      document.getElementById("event-title").innerText = "¡Juego Reiniciado!";
      document.getElementById("event-desc").innerText = "Lanza el dado para comenzar.";
      document.getElementById("options-container").innerHTML = "";
      document.getElementById("roll-btn").disabled = false;
      isMoving = false;
    }

    window.onload = createBoard;
    window.onresize = updateTokenPosition;
